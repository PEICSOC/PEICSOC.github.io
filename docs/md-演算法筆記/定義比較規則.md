# 定義比較規則


> 先不細談
定義st與st之間如何互相比較


```cpp
#include<bits/stdc++.h>
using namespace std;
#define nn "\n";

struct st{
    int x,y;
    bool operator== (const st& other) const {
        return x == other.x;
    }
    bool operator< (const st& other) const {
        return x < other.x;
    }
    bool operator> (const st& other) const {
        return x > other.x;
    }   
};

int main(){
    st a={3,5};
    st b={3,4};

    cout<<(a==b);

    return 0;
}
```
